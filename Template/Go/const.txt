func const_keyword(t)
    compile_const_type = ['bool', 'byte', 'sbyte', 'char', 'decimal', 'double', 'float', 'int', 'uint', 'nint', 'nuint', 'long', 'ulong', 'short', 'ushort', 'string']

    if compile_const_type | array.contains t
        ret 'var ' + t
    else
        ret 'var ' + t
    end
end

func format_const_value(type, value)
    if type == 'Duration'
        ret 'Duration(' + value + ')'
    else
        ret value
    end
end
