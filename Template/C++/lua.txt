namespace lua {

template <typename EnumType>
inline void enum_map(lua_State*)
{
    static_assert(sizeof(EnumType) == 0, "enum_map specialization is required for this type");
}

{{~ for enum in enums ~}}
template <>
inline void enum_map<{{ config.namespace | array.concat config.enum_namespace | array.add enum.name | array.join '::'}}>(lua_State* lua)
{
    lua_newtable(lua);
{{~ for prop in enum.props ~}}
    lua_pushinteger(lua, static_cast<lua_Integer>({{ config.namespace | array.concat config.enum_namespace | array.add enum.name | array.add prop.name | array.join '::'}}));
    lua_setfield(lua, -2, "{{ prop.name }}");
{{~ end ~}}
    lua_setglobal(lua, "{{ enum.name }}");
}

{{~ end ~}}

{
{{~ for enum in enums ~}}
    enum_map<{{ config.namespace | array.concat config.enum_namespace | array.add enum.name | array.join '::'}}>(lua);
{{~ end ~}}
}
