func load[T any](path string) T {
    file, err := os.Open(path)
    if err != nil {
        log.Fatal(err)
    }
    defer file.Close()

    bytes, err := io.ReadAll(file)
    if err != nil {
        log.Fatal(err)
    }

    var result T
    err = json.Unmarshal(bytes, &result)
    if err != nil {
        log.Fatal(err)
    }

    return result
}

func build{{ base_table | @upper_camel }}(path string) map[uint32]{{ base_table | @upper_camel }}Interface {
    concrete := load[map[uint32]{{ base_table | @upper_camel }}](path)
    result := make(map[uint32]{{ base_table | @upper_camel }}Interface)
    for key, value := range concrete {
        // Create a copy of the value to avoid pointer aliasing issues
        valueCopy := value
        result[key] = &valueCopy
    }
    return result
}
