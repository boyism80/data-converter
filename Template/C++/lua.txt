namespace lua {

template <typename EnumType>
inline void enum_map(lua_State*)
{
    static_assert(sizeof(EnumType) == 0, "enum_map specialization is required for this type");
}

{{~ for enum in enums ~}}
template <>
inline void enum_map<{{ ns.enum_root }}::{{ enum.name }}>(lua_State* lua)
{
    lua_newtable(lua);
{{~ for prop in enum.props ~}}
    lua_pushinteger(lua, static_cast<lua_Integer>({{ ns.enum_root }}::{{ enum.name }}::{{ prop.name }}));
    lua_setfield(lua, -2, "{{ prop.name }}");
{{~ end ~}}
    lua_setglobal(lua, "{{ enum.name }}");
}

{{~ end ~}}
inline static void map_enum(lua_State* lua)
{
{{~ for enum in enums ~}}
    enum_map<{{ ns.enum_root }}::{{ enum.name }}>(lua);
{{~ end ~}}
}

} // end of namespace {{ ns.root }}::lua
#endif
