template <typename K, typename V>
class kv_container : public container
{
public:
    using iterator          = std::unordered_map<K,V&>::iterator;
    using const_iterator    = std::unordered_map<K,V&>::const_iterator;

private:
    std::vector<std::unique_ptr<V>>     _ptrs;
    std::unordered_map<K,V&>            _pairs;
    const std::string                   _fname;

public:
    hook_funcs<V> hook;

protected:
    kv_container(std::string_view fname) : _fname(std::string(fname))
    { }

public:
    kv_container(const Json::Value& json)
    {
        this->load(json);
    }
    kv_container(const kv_container&) = delete;
    ~kv_container() = default;

private:
    void load(std::string_view fname)
    {
        auto fname_str = std::string(fname);
        Json::Value             json;
        std::ifstream           ifstream;
        ifstream.open(fname_str);
        if (ifstream.is_open() == false)
        {
            auto sstream = std::stringstream();
            sstream << "cannot find json file : " << fname_str;
            throw std::runtime_error(sstream.str());
        }

        ifstream >> json;
        ifstream.close();
        this->load(json);
    }
    void load(const Json::Value& json)
    {
        for (auto i = json.begin(); i != json.end(); i++)
        {
            auto ptr = std::unique_ptr<V>();
            if (this->hook.build != nullptr)
                ptr.reset(this->hook.build(*i));
            if (ptr == nullptr)
                ptr = {{ config.namespace | @namespace_access_cpp }}build<std::unique_ptr<V>>(*i);

            auto raw = ptr.get();
            this->_ptrs.push_back(std::move(ptr));
            this->_pairs.insert({ {{ config.namespace | @namespace_access_cpp }}build<K>(i.key()), *raw });

            if(this->hook.built != nullptr)
                this->hook.built(*raw);
        }
    }

public:
    void load()
    {
        this->load(std::string_view(this->_fname));
    }

    const std::string& fname() const
    {
        return this->_fname;
    }

    bool contains(const K& k) const
    {
        return this->_pairs.find(k) != this->_pairs.cend();
    }

    const V* find(const K& k) const
    {
        auto i = this->_pairs.find(k);
        if (i == this->_pairs.cend())
            return nullptr;

        return &i->second;
    }

    uint32_t size() const
    {
        return static_cast<uint32_t>(this->_pairs.size());
    }

    const V& operator [] (const K& k) const 
    {
        auto found = this->find(k);
        if(found == nullptr)
        {
            auto sstream = std::stringstream();
            if constexpr (std::is_enum_v<K>)
            {
                sstream << enum_tostring<K>(k);
            }
            else
            {
                sstream << k;
            }
            sstream << " does not exists in file " << this->_fname;
            throw std::runtime_error(sstream.str());
        }

        return *found;
    }

    iterator begin()
    {
        return this->_pairs.begin();
    }

    iterator end()
    {
        return this->_pairs.end();
    }

    const const_iterator begin() const
    {
        return this->_pairs.begin();
    }
    const const_iterator end() const
    {
        return this->_pairs.end();
    }
};

template <typename T>
class array_container : public container
{
public:
    class iterator;
    class const_iterator;

private:
    std::vector<std::unique_ptr<T>> _data;
    const std::string _fname;

public:
    hook_funcs<T> hook;

protected:
    array_container(std::string_view fname) : _fname(std::string(fname))
    { }

public:
    array_container(const Json::Value& json)
    {
        this->load(json);
    }
    array_container(const array_container&) = delete;
    ~array_container() = default;

private:
    void load(std::string_view fname)
    {
        auto fname_str = std::string(fname);
        Json::Value             json;
        std::ifstream           ifstream;
        ifstream.open(fname_str);
        if (ifstream.is_open() == false)
        {
            auto sstream = std::stringstream();
            sstream << "cannot find json file : " << fname_str;
            throw std::runtime_error(sstream.str());
        }

        Json::Reader            reader;
        if (reader.parse(ifstream, json) == false)
        {
            ifstream.close();

            auto sstream = std::stringstream();
            sstream << "cannot parse json file : " << fname_str;
            throw std::runtime_error(sstream.str());
        }

        ifstream.close();
        this->load(json);
    }
    void load(const Json::Value& json)
    {
        for (auto i = json.begin(); i != json.end(); i++)
        {
            auto ptr = std::unique_ptr<T>();
            if (this->hook.build != nullptr)
                ptr.reset(this->hook.build(*i));
            if (ptr == nullptr)
                ptr = {{ config.namespace | @namespace_access_cpp }}build<std::unique_ptr<T>>(*i);

            auto raw = ptr.get();
            this->_data.push_back(std::move(ptr));
            if(this->hook.built != nullptr)
                this->hook.built(*raw);
        }
    }

public:
    void load()
    {
        this->load(std::string_view(this->_fname));
    }

    const std::string& fname() const
    {
        return this->_fname;
    }

    const T* find(uint32_t i) const
    {
        if (i > this->_data.size() - 1)
            return nullptr;

        return this->_data.at(i).get();
    }

    uint32_t size() const
    {
        return static_cast<uint32_t>(this->_data.size());
    }

    const T& operator [] (uint32_t i) const
    {
        auto found = this->find(i);
        if (found == nullptr)
            throw std::out_of_range("out of range exception");

        return *found;
    }

public:
    iterator begin();
    iterator end();
    const const_iterator begin() const;
    const const_iterator end() const;
};

template <typename T>
class table_slot : public slot_base
{
private:
    std::shared_ptr<T>              _buf[2];
    std::atomic<int>                _active{0};
    std::atomic<std::shared_ptr<T>> _published;
    std::vector<std::shared_ptr<T>> _keep_alive;
    std::string                     _name;
    bool                            _loaded{false};

public:
    explicit table_slot(std::string_view name) :
        _name(std::string(name))
    {
        auto initial = std::make_shared<T>();
        this->_buf[0] = initial;
        this->_published.store(initial, std::memory_order_release);
        this->_keep_alive.push_back(initial);
    }

    table_slot(const table_slot&)            = delete;
    table_slot& operator=(const table_slot&) = delete;

    std::atomic<std::shared_ptr<T>>& hub()
    {
        return this->_published;
    }

    std::shared_ptr<T> published() const
    {
        return this->_published.load(std::memory_order_acquire);
    }

    std::string_view name() const override
    {
        return this->_name;
    }

    uint32_t size() const override
    {
        auto p = this->published();
        return p != nullptr ? p->size() : 0;
    }

    void load() override
    {
        if (this->_loaded)
        {
            this->reload();
            return;
        }

        auto cur = this->published();
        cur->load();
        this->_loaded = true;
    }

    void reload() override
    {
        auto cur  = this->published();
        auto next = std::make_shared<T>();
        // Copy factory hook only. Instance hooks (built) stay from T's constructor.
        if (cur != nullptr)
            next->hook.build = cur->hook.build;

        next->load();

        auto back = 1 - this->_active.load(std::memory_order_relaxed);
        this->_buf[back] = next;
        this->_active.store(back, std::memory_order_relaxed);
        this->_published.store(next, std::memory_order_release);
        this->_keep_alive.push_back(next);
        this->_loaded = true;
    }
};

template <typename T>
class table_view
{
private:
    std::atomic<std::shared_ptr<T>>* _hub = nullptr;
    std::shared_ptr<T>               _p;

public:
    explicit table_view(table_slot<T>& slot) :
        _hub(&slot.hub()),
        _p(slot.published())
    { }

    // auto snap = table::spell;  // pins published generation
    table_view(const table_view& rhs) :
        _hub(nullptr),
        _p(rhs._hub != nullptr ? rhs._hub->load(std::memory_order_acquire) : rhs._p)
    { }

    table_view(table_view&&)                 = delete;
    table_view& operator=(const table_view&) = delete;
    table_view& operator=(table_view&&)      = delete;

    std::shared_ptr<T> get() const
    {
        if (this->_hub != nullptr)
            return this->_hub->load(std::memory_order_acquire);
        return this->_p;
    }

    // Root: thread_local pin of published (safe for -> method calls).
    // Snapshot: use pinned generation.
    T* operator->() const
    {
        if (this->_hub != nullptr)
        {
            thread_local std::shared_ptr<T> tls;
            tls = this->_hub->load(std::memory_order_acquire);
            return tls.get();
        }
        assert(this->_p != nullptr);
        return this->_p.get();
    }

    T& operator*() const
    {
        return *this->operator->();
    }

    // Prefer snapshot: auto snap = table::spell; snap[id]
    // Root uses thread_local pin — do not keep returned refs across another root [].
    template <typename K>
    decltype(auto) operator[](const K& k) const
    {
        if (this->_hub != nullptr)
        {
            thread_local std::shared_ptr<T> tls;
            tls = this->_hub->load(std::memory_order_acquire);
            return (*tls)[k];
        }
        assert(this->_p != nullptr);
        return (*this->_p)[k];
    }

    auto begin() const
    {
        auto p = this->pinned();
        return p->begin();
    }

    auto end() const
    {
        auto p = this->pinned();
        return p->end();
    }

    auto begin()
    {
        auto p = this->pinned();
        return p->begin();
    }

    auto end()
    {
        auto p = this->pinned();
        return p->end();
    }

    uint32_t size() const
    {
        return this->pinned()->size();
    }

private:
    std::shared_ptr<T> pinned() const
    {
        if (this->_hub != nullptr)
        {
            thread_local std::shared_ptr<T> tls;
            tls = this->_hub->load(std::memory_order_acquire);
            return tls;
        }
        assert(this->_p != nullptr);
        return this->_p;
    }
};
