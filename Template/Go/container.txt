func readJsonFile(path string) (json.RawMessage, error) {
	jsonFile, err := os.Open(path)
	if err != nil {
		return nil, err
	}
	defer jsonFile.Close()

	byteValue, _ := io.ReadAll(jsonFile)
	return json.RawMessage(byteValue), nil
}

// LoadAll loads all tables in parallel
func LoadAll(callback func(float64)) {
	count := len(Containers)
	completed := int32(0)
	wg := sync.WaitGroup{}
	wg.Add(count)

	for _, container := range Containers {
		container := container // Create a new variable in a new scope to avoid closure issues
		go func() {
			defer wg.Done()
			if err := container.Load(); err != nil {
				fmt.Printf("Error loading container: %v\n", err)
				return
			}
			newCompleted := atomic.AddInt32(&completed, 1)
			callback(float64(newCompleted) / float64(count))
		}()
	}
	wg.Wait()
}
