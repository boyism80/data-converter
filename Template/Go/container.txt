func readJsonFile(path string) (json.RawMessage, error) {
	jsonFile, err := os.Open(path)
	if err != nil {
		return nil, err
	}
	defer jsonFile.Close()

	byteValue, _ := io.ReadAll(jsonFile)
	return json.RawMessage(byteValue), nil
}

func (container *Container) Load(callback func(float64)) {
	type LoadTask struct {
		Path string
		Func func(json.RawMessage) error
	}

	tasks := []LoadTask{
{{~ for table in tables | array.filter @containable ~}}
		{
			Path: "json/{{ table.name }}.json",
			Func: func(data json.RawMessage) error {
				builder := {{ table.name | @upper_camel }}ContainerBuilder{
					Hook: container.{{ table.name | @upper_camel }}Hook,
				}
				result, err := builder.Build(data)
				if err != nil {
					return err
				}
				container.{{ table.name | @upper_camel }} = result
				return nil
			},
		},
{{~ end ~}}
	}

	count := len(tasks)
	completed := int32(0)
	wg := sync.WaitGroup{}
	wg.Add(count)

	for _, task := range tasks {
		task := task // Create a new task variable in a new scope to avoid closure issues
		go func() {
			defer wg.Done()
			jsonData, err := readJsonFile(task.Path)
			if err != nil {
				fmt.Printf("Error reading file %s: %v\n", task.Path, err)
				return
			}
			if err := task.Func(jsonData); err != nil {
				fmt.Printf("Error processing file %s: %v\n", task.Path, err)
				return
			}
			newCompleted := atomic.AddInt32(&completed, 1)
			callback(float64(newCompleted) / float64(count))
		}()
	}
	wg.Wait()
}
